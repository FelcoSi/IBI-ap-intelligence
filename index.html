  <script>

    let globalDataRaw = [];

    let isYearToDateMode = false;

    let locationChartInstance = null;

    let categoryChartInstance = null;


    window.addEventListener('DOMContentLoaded', () => {

      initializeDashboardLifecycle();

    });


    function initializeDashboardLifecycle() {

      const storedUrl = localStorage.getItem('ibi_dashboard_api_url');

      if (!storedUrl) {

        document.getElementById('setupGate').classList.remove('hidden');

        document.getElementById('dashboardCore').classList.add('hidden');

      } else {

        document.getElementById('setupGate').classList.add('hidden');

        document.getElementById('dashboardCore').classList.remove('hidden');

        fetchPayloadFromSource(storedUrl);

      }

    }


    function saveConfig() {

      const inputUrl = document.getElementById('apiUrlInput').value.trim();

      if (inputUrl.includes('script.google.com')) {

        localStorage.setItem('ibi_dashboard_api_url', inputUrl);

        initializeDashboardLifecycle();

      } else {

        alert('Invalid link. Please supply a valid Google Apps Script Web App Deployment URL.');

      }

    }


    function clearConfig() {

      localStorage.removeItem('ibi_dashboard_api_url');

      location.reload();

    }


    function fetchPayloadFromSource(apiUrl) {

      // Append streaming parameter payload descriptor to hook the script endpoint gateway

      fetch(`${apiUrl}?stream=live_dashboard`)

        .then(response => response.json())

        .then(data => {

          globalDataRaw = data;

          processAndRenderDashboardMatrix();

        })

        .catch(err => {

          alert("Database connection timeout or formatting error. Check deployment permissions.");

          clearConfig();

        });

    }


    function toggleViewMode() {

      isYearToDateMode = !isYearToDateMode;

      document.getElementById('viewToggle').textContent = isYearToDateMode ? "SWITCH TO MONTHLY VIEW" : "SWITCH TO YTD VIEW";

      document.getElementById('contextBadge').textContent = isYearToDateMode ? "Active Dashboard Scope: Year-To-Date (YTD) Summary" : "Active Dashboard Scope: Current Month (MTD)";

      processAndRenderDashboardMatrix();

    }


    function processAndRenderDashboardMatrix() {

      const today = new Date();

      const currentMonth = today.getMonth();

      const currentYear = today.getFullYear();


      // Filter data depending on chosen scope mode

      let filteredRecords = globalDataRaw.filter(record => {

        if (!record.invoice_date) return false;

        let rDate = new Date(record.invoice_date);

        if (isNaN(rDate.getTime())) return false;


        if (isYearToDateMode) {

          return rDate.getFullYear() === currentYear; // Match current calendar year

        } else {

          return rDate.getMonth() === currentMonth && rDate.getFullYear() === currentYear; // Strict active month mapping

        }

      });


      // Reset KPI counters

      let total = 0, approved = 0, pending = 0, pastDue = 0;

      let locationAggregator = {};

      let categoryAggregator = {};


      filteredRecords.forEach(rec => {

        let amt = parseFloat(rec.amount) || 0;

        total += amt;

        

        let stat = String(rec.status).trim();

        if (stat === "Approved" || stat === "Paid") approved += amt;

        if (stat.includes("Pending")) pending += amt;

        if (stat === "Past Due") pastDue += amt;


        // Compile location volume arrays

        let loc = rec.location ? rec.location.trim() : "Unassigned";

        locationAggregator[loc] = (locationAggregator[loc] || 0) + amt;


        // Compile category volume arrays

        let cat = rec.category ? rec.category.trim() : "Unassigned";

        categoryAggregator[cat] = (categoryAggregator[cat] || 0) + amt;

      });


      // Update values in the UI DOM layout

      document.getElementById('kpiTotal').textContent = formatCurrency(total);

      document.getElementById('kpiApproved').textContent = formatCurrency(approved);

      document.getElementById('kpiPending').textContent = formatCurrency(pending);

      document.getElementById('kpiPastDue').textContent = formatCurrency(pastDue);


      renderCharts(locationAggregator, categoryAggregator);

    }


    function formatCurrency(val) {

      return '$' + val.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });

    }


    function renderCharts(locationData, categoryData) {

      // Destruction markers to clean memory instances before drawing charts

      if (locationChartInstance) locationChartInstance.destroy();

      if (categoryChartInstance) categoryChartInstance.destroy();


      Chart.defaults.color = '#94a3b8';

      Chart.defaults.font.family = '-apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif';


      // Draw Location Chart (Bar Visuals)

      const locCtx = document.getElementById('locationChart').getContext('2d');

      locationChartInstance = new Chart(locCtx, {

        type: 'bar',

        data: {

          labels: Object.keys(locationData),

          datasets: [{

            label: 'Total Expenses',

            data: Object.values(locationData),

            backgroundColor: '#3b82f6',

            borderRadius: 6,

            borderSkipped: false

          }]

        },

        options: {

          responsive: true,

          maintainAspectRatio: false,

          plugins: { legend: { display: false } },

          scales: {

            x: { grid: { display: false } },

            y: { grid: { color: '#334155' } }

          }

        }

      });


      // Draw Category Chart (Doughnut/Pie Slices Matrix)

      const catCtx = document.getElementById('categoryChart').getContext('2d');

      categoryChartInstance = new Chart(catCtx, {

        type: 'doughnut',

        data: {

          labels: Object.keys(categoryData),

          datasets: [{

            data: Object.values(categoryData),

            backgroundColor: ['#6366f1', '#10b981', '#f59e0b', '#ec4899', '#8b5cf6', '#3b82f6', '#64748b'],

            borderWidth: 0

          }]

        },

        options: {

          responsive: true,

          maintainAspectRatio: false,

          plugins: {

            legend: { position: 'bottom', labels: { boxWidth: 12, padding: 15 } }

          }

        }

      });

    }
